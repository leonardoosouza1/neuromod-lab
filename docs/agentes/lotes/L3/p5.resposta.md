PREDICAO: mortos ≈0.70 (de 0.74), corridas ≈17.0 (de 16.3, ticks livres viram corrida), refeições ≈7.0 (quase igual, comida seguir prioridade 1 garante 1º slot), abastecidas ≈5.5 (de 7.5, mais perto do campeão 4.3 pois fuel só entra quando sobra slot), tempo livre no raio de serviço cai de 42.8% pra ~30% (mais perto do campeão 26.2%), dia médio morte ≈6.0.

Justificativa: dado batido — tempo LIVRE dentro do raio de serviço é 42.8% na r7_1 vs 26.2% no campeão. Ordem comida>agua>fuel já é ótima por taxa de degradação (fome .32/tick > água .08–.2/tick > combustível 0/tick parado, já que so' gasta "andando" — reordenar isso pioraria). O gargalo real é QUANTOS TICKS fica: hoje o posto deixa comprar as 3 coisas em sequência (até 3 ticks parado) sempre que as 3 são desejadas ao mesmo tempo. Limito a 2 compras por parada (LIMITE_PARADA=2): garante sempre a compra de maior prioridade (comida, depois água OU combustível conforme o score), e empurra a 3ª pra uma parada futura — combustível é o que mais cede porque não decai parado, então adiar é seguro. Isso aproxima o padrão do campeão (4.3 abastecidas/vida, vs 7.5 hoje: menos paradas completas, mais paradas dedicadas).

import torch


def fabrica():
    class PoliticaConstrucaoR7A1OrdemPosto:
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.comeu_ultimo_tick = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.ticks_parada = torch.zeros(m.B, dtype=torch.long, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # trabalho continuo: sempre ao alvo. NUNCA argmin de d_posto.
            mover = torch.argmin(F[:, :, 0], dim=1)
            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)

            score = torch.full((B, S), -1e9, device=device, dtype=dtype)

            quer_agua = e_agua < 0.8
            quer_comida = e_fome > 0.15
            quer_fuel = e_fuel < 0.95

            # ORDEM DE COMPRA NO POSTO: comida > agua > fuel (inalterado — validado pela taxa de
            # degradacao: fome .32/tick > agua .08-.2/tick > combustivel 0/tick parado, ja que so'
            # gasta "andando"; reordenar pioraria porque combustivel nao piora enquanto parado).
            score[:, 7] = torch.where(no_posto & quer_comida, 30.0 + (e_fome - 0.15), score[:, 7])
            score[:, 5] = torch.where(no_posto & quer_agua, 20.0 + (0.8 - e_agua), score[:, 5])

            idx_pref = torch.where(
                dinh > 40,
                torch.full((B,), 3, device=device, dtype=torch.long),
                torch.full((B,), 1, device=device, dtype=torch.long),
            )
            disp_pref = disp.gather(1, idx_pref.unsqueeze(1)).squeeze(1)
            idx_fuel = torch.where(disp_pref, idx_pref, torch.zeros_like(idx_pref))
            disp_fuel = torch.where(disp_pref, disp_pref, disp[:, 0])
            val_fuel = torch.where(
                no_posto & quer_fuel & disp_fuel,
                10.0 + (1.0 - e_fuel),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_fuel.unsqueeze(1), val_fuel.unsqueeze(1))

            # ALAVANCA QUANTOS TICKS FICA: no maximo 2 compras por parada no posto (medido: r7_1
            # fica 42.8% do tempo livre parado no raio de servico vs 26.2% do campeao — sequencia
            # de ate 3 compras por parada e' o gargalo). A 3a necessidade (sempre combustivel, por
            # ser a de menor prioridade e a unica que nao degrada parado) fica pra uma parada
            # futura, forcando o taxi a sair (mover) e trabalhar em vez de ficar parado comprando.
            LIMITE_PARADA = 2
            pode_comprar = self.ticks_parada < LIMITE_PARADA

            melhor = torch.argmax(score, dim=1)
            comprar = no_posto & pode_comprar & (score.gather(1, melhor.unsqueeze(1)).squeeze(1) > -1e8)

            # segunda refeicao: se comeu no tick anterior e ainda ta com fome e tem grana, come de novo
            segunda = no_posto & pode_comprar & self.comeu_ultimo_tick & (e_fome > 0.15) & (dinh > 30) & disp[:, 7]

            # ALAVANCA NOITE: posto fecha de noite (dados: r7_1 tem quase 0 refeicoes 1h-5h vs
            # campeao evoluido 70-125/h a noite toda, pois o campeao come em ponto de comida, nao
            # so na loja do posto). De dia a compra continua na loja do posto (score[:,7] acima,
            # inalterado). De noite (ou fome > .6 mesmo de dia) desvia pro ponto de comida mais
            # perto com raio maior (25, contra 20 antes) pois de noite o ponto costuma estar mais
            # longe. Nunca desvia com passageiro a bordo (~tem_pax) — corrida termina primeiro.
            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            # emergencia fora do posto: so' se ja esta ao lado
            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            # REPARO: (1) vida baixa com histerese .5/.8, (2) calor alto sem passageiro
            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.8) & (dinh > 10)
            quer_calor = (e_calor > 0.75) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            # prioridade (menor pra maior, ultima escrita vence):
            # mover base < buscar comida de longe < detour oficina < compra posto < compra oficina
            # < emergencia agua/comida < segunda refeicao
            acao = mover.clone()
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)
            acao = torch.where(segunda, torch.full_like(acao, G + 7), acao)

            self.comeu_ultimo_tick = acao == (G + 7)

            fez_compra_posto = comprar | segunda
            self.ticks_parada = torch.where(
                no_posto,
                self.ticks_parada + fez_compra_posto.long(),
                torch.zeros_like(self.ticks_parada),
            )

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaConstrucaoR7A1OrdemPosto
