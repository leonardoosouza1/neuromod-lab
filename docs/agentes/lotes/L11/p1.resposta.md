PREDICAO: quase empate — mortos ≈ 0.55–0.58, corridas ≈ 3.2–3.3/dia vivo (base 3.28). Refeições devem cair de 8.4 para ~6–7 (menos compras espúrias no posto), mas fome como causa de morte deve ficar igual ou subir 1–2 pp (47%→~48–49%), como em L10 p1 (0.57, refeições 6.8, fome 49%). Justificativa dos números: a REFEICAO custa só 2×pmul (~0.6–2.6 de caixa) — irrisório frente aos 15–60 de um tanque cheio — então essa alavanca não ataca o gargalo de caixa (H34); o ganho real só pode vir de TICKS: em `e_fome>0.15` o táxi compra comida quase toda vez que passa no posto (fome sobe ~0.0032/tick × ~57 ticks entre corridas ≈ 0.18 de folga, sempre acima de 0.15), gastando 1 tick de parada sem necessidade real. Subindo para `0.30` — o valor que o próprio consolidador citou para PARADA MÍNIMA ("comida se fome > .3") — ainda fica bem abaixo da mediana real de quem come (0.637 do campeão, 0.701 da política explícita) e acima do pior caso observado (p10 0.211–0.384), então não deveria empurrar mortes por fome. Mantive intocados os limiares "fora" (busca noturna `e_noite>0.5`, dia `e_fome>0.6`, emergência `e_fome>0.85`) porque são a rede de segurança validada (+2pp de comer à noite) e L10 p1 sugere que mexer neles é que piora a nutrição — a alavanca aqui é só o limiar de COMPRA no posto.

```python
import torch


def fabrica():
    class PoliticaL11Comida(object):
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.comeu_ultimo_tick = torch.zeros(m.B, dtype=torch.bool, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # ALAVANCA RUA (herdada de p7, intacta): score de mover ao alvo pondera d_alvo +
            # peso*custo_frente. peso pequeno de proposito: L1 p7 usou peso=5.0 e a penalidade
            # (5*2=10) dominou o d_alvo (diff maxima ~2), paralisando o taxi (0.99, 1.7 corridas).
            # Aqui peso=.15 da penalidade maxima .15*2=.3, so' desempata direcoes quase
            # equivalentes sem nunca virar a decisao principal de ir ao alvo.
            PESO_RUA = 0.15
            mover = torch.argmin(F[:, :, 0] + PESO_RUA * F[:, :, 6], dim=1)
            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)

            score = torch.full((B, S), -1e9, device=device, dtype=dtype)

            quer_agua = e_agua < 0.50
            # ALAVANCA COMIDA (unica mudanca vs base L6 p4): limiar de compra no posto
            # sobe de .15 para .30. A REFEICAO custa so' 2*pmul (~0.6-2.6 de caixa) — nao
            # e' isso que disputa caixa com gasolina (H34), entao o ganho aqui e' de
            # TICKS, nao de dinheiro: com .15, fome (+.0032/tick normalizado) quase
            # sempre passa do limiar entre duas visitas ao posto (mediana de 57 ticks
            # entre corridas -> +-0.18 de fome), fazendo o taxi comprar comida toda vez
            # que passa por la', gastando 1 tick de parada que podia ir pra corrida. .30
            # ainda fica bem abaixo de onde o campeao (mediana .637, p10 .384) e a
            # politica explicita (mediana .701, p10 .211) realmente comem, entao nao
            # deveria empurrar mortes por fome (so' reduz compra precoce/redundante).
            LIMIAR_COMIDA = 0.30
            quer_comida = e_fome > LIMIAR_COMIDA
            quer_fuel = e_fuel < 0.95

            # ORDEM DE COMPRA NO POSTO: comida > agua > fuel (inalterado da base).
            score[:, 7] = torch.where(no_posto & quer_comida, 30.0 + (e_fome - LIMIAR_COMIDA), score[:, 7])

            # ALAVANCA AGUA (herdada de L10 p2, intacta): tier padrao 50L, sobe pra 100L
            # so' com e_agua<0.25 E dinh>40 (nao disputa caixa com comida/gasolina).
            idx_pref_agua = torch.where(
                (e_agua < 0.25) & (dinh > 40.0),
                torch.full((B,), 6, device=device, dtype=torch.long),
                torch.full((B,), 5, device=device, dtype=torch.long),
            )
            disp_pref_agua = disp.gather(1, idx_pref_agua.unsqueeze(1)).squeeze(1)
            idx_agua = torch.where(disp_pref_agua, idx_pref_agua, torch.full_like(idx_pref_agua, 5))
            disp_agua = torch.where(disp_pref_agua, disp_pref_agua, disp[:, 5])
            val_agua = torch.where(
                no_posto & quer_agua & disp_agua,
                20.0 + (0.50 - e_agua),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_agua.unsqueeze(1), val_agua.unsqueeze(1))

            # ALAVANCA GASOLINA (herdada de L6 p4, intacta): tier = min(tier_por_caixa,
            # tier_por_necessidade), nunca compra mais litro do que cabe no tanque nem
            # mais do que o caixa permite.
            liters_falta = (1.0 - e_fuel) * 100.0
            idx_necessidade = torch.where(
                liters_falta <= 25.0,
                torch.full((B,), 0, device=device, dtype=torch.long),
                torch.where(
                    liters_falta <= 50.0,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.where(
                        liters_falta <= 75.0,
                        torch.full((B,), 2, device=device, dtype=torch.long),
                        torch.full((B,), 3, device=device, dtype=torch.long),
                    ),
                ),
            )
            idx_caixa = torch.where(
                dinh > 50,
                torch.full((B,), 3, device=device, dtype=torch.long),
                torch.where(
                    dinh > 25,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.full((B,), 0, device=device, dtype=torch.long),
                ),
            )
            idx_pref = torch.minimum(idx_caixa, idx_necessidade)
            disp_pref = disp.gather(1, idx_pref.unsqueeze(1)).squeeze(1)
            idx_fuel = torch.where(disp_pref, idx_pref, torch.zeros_like(idx_pref))
            disp_fuel = torch.where(disp_pref, disp_pref, disp[:, 0])
            val_fuel = torch.where(
                no_posto & quer_fuel & disp_fuel,
                10.0 + (1.0 - e_fuel),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_fuel.unsqueeze(1), val_fuel.unsqueeze(1))

            melhor = torch.argmax(score, dim=1)
            comprar = no_posto & (score.gather(1, melhor.unsqueeze(1)).squeeze(1) > -1e8)

            # segunda compra na mesma parada usa o mesmo limiar novo (.30) — coerente
            # com a compra principal, so' dispara se a fome continuar alta apos comer.
            segunda = no_posto & self.comeu_ultimo_tick & (e_fome > LIMIAR_COMIDA) & (dinh > 30) & disp[:, 7]

            # limiares "fora" (busca noturna, busca diurna, emergencia) intocados: sao a
            # rede de seguranca ja validada (+2pp de comer a noite) e mexer neles em L10
            # p1 empurrou fome-causa de 47% pra 49% sem ganho — nao repetir.
            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.8) & (dinh > 10)
            quer_calor = (e_calor > 0.75) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            acao = mover.clone()
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)
            acao = torch.where(segunda, torch.full_like(acao, G + 7), acao)

            self.comeu_ultimo_tick = acao == (G + 7)

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaL11Comida
```
