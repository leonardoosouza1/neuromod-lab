PREDICAO: mortos ≈ 0.53–0.57 (leve melhora a empate sobre a base 0.56 — o mecanismo usa uma informação nova, preço local do combustível, que nunca entrou no limiar de caixa da gasolina em nenhum dos 4 testes anteriores da alavanca; mas o teto de litros continua travado por `idx_necessidade`, então o efeito é só realocação de caixa entre gasolina/comida/água, não mudança de movimento — não espero salto). corridas por dia vivo ≈ 3.20–3.35 (perto da base 3.28: a alavanca não toca trabalho/lugar, só o tamanho da compra de combustível).

```python
import torch


def fabrica():
    class PoliticaGasolinaPreco(object):
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.comeu_ultimo_tick = torch.zeros(m.B, dtype=torch.bool, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # ALAVANCA RUA (herdada de p7, intacta): score de mover ao alvo pondera d_alvo +
            # peso*custo_frente. peso pequeno de proposito: L1 p7 usou peso=5.0 e a penalidade
            # (5*2=10) dominou o d_alvo (diff maxima ~2), paralisando o taxi (0.99, 1.7 corridas).
            # Aqui peso=.15 da penalidade maxima .15*2=.3, so' desempata direcoes quase
            # equivalentes sem nunca virar a decisao principal de ir ao alvo.
            PESO_RUA = 0.15
            mover = torch.argmin(F[:, :, 0] + PESO_RUA * F[:, :, 6], dim=1)
            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)

            score = torch.full((B, S), -1e9, device=device, dtype=dtype)

            quer_agua = e_agua < 0.50
            quer_comida = e_fome > 0.15
            quer_fuel = e_fuel < 0.95

            # ORDEM DE COMPRA NO POSTO: comida > agua > fuel (inalterado da base).
            score[:, 7] = torch.where(no_posto & quer_comida, 30.0 + (e_fome - 0.15), score[:, 7])

            # ALAVANCA AGUA (herdada da base L10 p2, intacta): gatilho e tier padrao batem
            # com os dados medidos (campeao bebe com e_agua mediana 0.545); o unico ajuste
            # ja incorporado e' nao encher 100L se dinh<=40, seguindo o mesmo padrao que
            # funcionou em GASOLINA (nunca comprar mais do que cabe/precisa quando o caixa
            # esta apertado). Mantido intacto porque a tarefa deste lote e' so' GASOLINA.
            idx_pref_agua = torch.where(
                (e_agua < 0.25) & (dinh > 40.0),
                torch.full((B,), 6, device=device, dtype=torch.long),
                torch.full((B,), 5, device=device, dtype=torch.long),
            )
            disp_pref_agua = disp.gather(1, idx_pref_agua.unsqueeze(1)).squeeze(1)
            idx_agua = torch.where(disp_pref_agua, idx_pref_agua, torch.full_like(idx_pref_agua, 5))
            disp_agua = torch.where(disp_pref_agua, disp_pref_agua, disp[:, 5])
            val_agua = torch.where(
                no_posto & quer_agua & disp_agua,
                20.0 + (0.50 - e_agua),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_agua.unsqueeze(1), val_agua.unsqueeze(1))

            # ALAVANCA GASOLINA (UNICA MUDANCA deste lote): idx_necessidade continua
            # travando o teto em litros_falta (nunca compra mais do que cabe/precisa —
            # isso refuta H22, top-up voluntario, e nao mexemos nisso). O que muda e'
            # idx_caixa: antes os limiares de dinheiro (>50 -> 100L, >25 -> 50L) eram
            # fixos, como se o litro custasse sempre o mesmo em qualquer distrito. Mas o
            # contrato diz preco_L = .3 x pmul do distrito, e os pmul medidos no mapa de
            # teste vao de .50 (periferia) a 1.30 (centro) -- o mesmo tanque de 100L custa
            # ~2.6x mais no centro do que na periferia. H34 (gargalo e' caixa: base
            # abastece com dinh 20.8 contra 37.1 do campeao, compra 50L contra 100L) nunca
            # foi atacado por esse angulo em nenhum dos 4 testes anteriores da alavanca
            # (L5/L6/L8/L10 p4) -- todos mexeram so' no VALOR do limiar, nao em torna-lo
            # sensivel ao preco local.
            #
            # e_preco_aqui le' preco_frente na opcao 'ficar' (G-1 = indice 8, que nao anda
            # e portanto reflete o distrito ONDE o taxi esta agora -- mesma convencao ja
            # usada no arquivo para d_alvo/d_agua/d_comida "aqui").
            #
            # fator_preco = preco_aqui / 0.85 (0.85 = pmul do distrito residencial, o
            # "neutro" do mapa de teste, que e' o cenario em que os limiares originais
            # 50/25 foram calibrados). Na periferia (pmul .50, fator~.59) o limiar de
            # 100L cai para ~30 -> tanque cheio custa pouco, sobra caixa pra comida/agua
            # depois (o proprio racional do H26 sobre a AGUA, aplicado aqui a gasolina).
            # No centro (pmul 1.30, fator~1.53) o limiar sobe para ~76 -> so enche o
            # tanque com bastante caixa sobrando; caso contrario compra so' 50L e guarda
            # o resto para comida (evita repetir H22: nunca enche o tanque por vontade,
            # so' quando o caixa realmente cobre com folga).
            e_preco_aqui = F[:, G - 1, 7]
            fator_preco = torch.clamp(e_preco_aqui / 0.85, 0.5, 1.8)
            limiar_100 = 50.0 * fator_preco
            limiar_50 = 25.0 * fator_preco

            liters_falta = (1.0 - e_fuel) * 100.0
            idx_necessidade = torch.where(
                liters_falta <= 25.0,
                torch.full((B,), 0, device=device, dtype=torch.long),
                torch.where(
                    liters_falta <= 50.0,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.where(
                        liters_falta <= 75.0,
                        torch.full((B,), 2, device=device, dtype=torch.long),
                        torch.full((B,), 3, device=device, dtype=torch.long),
                    ),
                ),
            )
            idx_caixa = torch.where(
                dinh > limiar_100,
                torch.full((B,), 3, device=device, dtype=torch.long),
                torch.where(
                    dinh > limiar_50,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.full((B,), 0, device=device, dtype=torch.long),
                ),
            )
            idx_pref = torch.minimum(idx_caixa, idx_necessidade)
            disp_pref = disp.gather(1, idx_pref.unsqueeze(1)).squeeze(1)
            idx_fuel = torch.where(disp_pref, idx_pref, torch.zeros_like(idx_pref))
            disp_fuel = torch.where(disp_pref, disp_pref, disp[:, 0])
            val_fuel = torch.where(
                no_posto & quer_fuel & disp_fuel,
                10.0 + (1.0 - e_fuel),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_fuel.unsqueeze(1), val_fuel.unsqueeze(1))

            melhor = torch.argmax(score, dim=1)
            comprar = no_posto & (score.gather(1, melhor.unsqueeze(1)).squeeze(1) > -1e8)

            segunda = no_posto & self.comeu_ultimo_tick & (e_fome > 0.15) & (dinh > 30) & disp[:, 7]

            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.8) & (dinh > 10)
            quer_calor = (e_calor > 0.75) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            acao = mover.clone()
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)
            acao = torch.where(segunda, torch.full_like(acao, G + 7), acao)

            self.comeu_ultimo_tick = acao == (G + 7)

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaGasolinaPreco
```
