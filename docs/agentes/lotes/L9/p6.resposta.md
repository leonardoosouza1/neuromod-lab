Vou implementar a alavanca "livre = caçar passageiro" (H30), já que as notas do lote deixam explícito que NOITE/descanso está aposentada e o slot deve virar a lógica de sair do raio do serviço quando não há compra pendente. A única mudança real em relação à base é penalizar a opção "ficar" quando o táxi está livre (sem passageiro), sem pendência de compra e dentro do raio de um serviço — isso força o "mover" (que já mira o alvo/passageiro mais próximo via d_alvo) a escolher uma direção real em vez de ociosidade.

PREDICAO: mortos ≈0.52–0.55 (melhora modesta sobre a base 0.56 — H30 previu -0.05 a -0.08 via +1.5 corridas, mas calibro pra baixo porque L7/L8 p8 já tentaram algo parecido em "LIVRE" e só empataram em 0.60; aqui a alavanca é mais cirúrgica — só penaliza 'ficar' quando não há pendência nenhuma — então espero corridas ≈19.3–19.7 (não abaixo do piso de 18.5) e causas de morte quase inalteradas (fome ~46%, dano ~49%, sede ~20%), já que não mexo em nenhum limiar de compra.

```python
import torch


def fabrica():
    class PoliticaL9Livre:
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.comeu_ultimo_tick = torch.zeros(m.B, dtype=torch.bool, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # ALAVANCA RUA (herdada de p7, intacta): peso pequeno so' desempata direcoes
            # quase equivalentes, nunca vira a decisao principal (L1 p7 com peso=5.0
            # paralisou o taxi: 0.99, 1.7 corridas).
            PESO_RUA = 0.15
            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)

            score = torch.full((B, S), -1e9, device=device, dtype=dtype)

            quer_agua = e_agua < 0.50
            quer_comida = e_fome > 0.15
            quer_fuel = e_fuel < 0.95

            # ORDEM DE COMPRA NO POSTO: comida > agua > fuel (inalterado da base).
            score[:, 7] = torch.where(no_posto & quer_comida, 30.0 + (e_fome - 0.15), score[:, 7])

            idx_pref_agua = torch.where(
                e_agua < 0.25,
                torch.full((B,), 6, device=device, dtype=torch.long),
                torch.full((B,), 5, device=device, dtype=torch.long),
            )
            disp_pref_agua = disp.gather(1, idx_pref_agua.unsqueeze(1)).squeeze(1)
            idx_agua = torch.where(disp_pref_agua, idx_pref_agua, torch.full_like(idx_pref_agua, 5))
            disp_agua = torch.where(disp_pref_agua, disp_pref_agua, disp[:, 5])
            val_agua = torch.where(
                no_posto & quer_agua & disp_agua,
                20.0 + (0.50 - e_agua),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_agua.unsqueeze(1), val_agua.unsqueeze(1))

            # ALAVANCA GASOLINA (herdada de L6 p4, intacta): idx_pref = min(tier_por_caixa,
            # tier_por_necessidade), nunca compra mais litro do que cabe no tanque.
            liters_falta = (1.0 - e_fuel) * 100.0
            idx_necessidade = torch.where(
                liters_falta <= 25.0,
                torch.full((B,), 0, device=device, dtype=torch.long),
                torch.where(
                    liters_falta <= 50.0,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.where(
                        liters_falta <= 75.0,
                        torch.full((B,), 2, device=device, dtype=torch.long),
                        torch.full((B,), 3, device=device, dtype=torch.long),
                    ),
                ),
            )
            idx_caixa = torch.where(
                dinh > 50,
                torch.full((B,), 3, device=device, dtype=torch.long),
                torch.where(
                    dinh > 25,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.full((B,), 0, device=device, dtype=torch.long),
                ),
            )
            idx_pref = torch.minimum(idx_caixa, idx_necessidade)
            disp_pref = disp.gather(1, idx_pref.unsqueeze(1)).squeeze(1)
            idx_fuel = torch.where(disp_pref, idx_pref, torch.zeros_like(idx_pref))
            disp_fuel = torch.where(disp_pref, disp_pref, disp[:, 0])
            val_fuel = torch.where(
                no_posto & quer_fuel & disp_fuel,
                10.0 + (1.0 - e_fuel),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_fuel.unsqueeze(1), val_fuel.unsqueeze(1))

            melhor = torch.argmax(score, dim=1)
            comprar = no_posto & (score.gather(1, melhor.unsqueeze(1)).squeeze(1) > -1e8)

            segunda = no_posto & self.comeu_ultimo_tick & (e_fome > 0.15) & (dinh > 30) & disp[:, 7]

            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.8) & (dinh > 10)
            quer_calor = (e_calor > 0.75) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            # ALAVANCA DO LOTE (unica mudanca vs a base): NOITE/descanso foi aposentada
            # (0.74-0.78 em 3 lotes, corta renda sem tocar na fome noturna). No lugar,
            # ataco a ociosidade medida (H30): o campeao passa so' 26.2% do tempo livre
            # parado no raio de um servico, a melhor politica ate agora passa 42.8% —
            # 17 pp de tempo parado perto de posto/oficina sem comprar nada, que no
            # campeao viram corrida. 'mover' ja mira o alvo mais proximo via d_alvo
            # (coluna 0 de F), mas a opcao 'ficar' (indice G-1) pode empatar/ganhar
            # quando o taxi esta parado num servico sem nenhuma pendencia de compra,
            # porque custo_frente penaliza sair do asfalto do posto. Aqui penalizo
            # 'ficar' nesse caso especifico (livre, sem passageiro, sem pendencia,
            # dentro do raio de posto/oficina) para forcar a saida em direcao ao
            # passageiro mais proximo, sem mexer em nenhum limiar de compra/agua/
            # comida/oficina/gasolina ja calibrado nos lotes anteriores.
            sem_pendencia = ~(quer_agua | quer_comida | quer_fuel | quer_oficina)
            no_servico = no_posto | no_oficina
            livre_ocioso = (~tem_pax) & sem_pendencia & no_servico

            d_score = F[:, :, 0] + PESO_RUA * F[:, :, 6]
            penalidade_ficar = torch.where(livre_ocioso, torch.full((B,), 50.0, device=device, dtype=dtype), torch.zeros(B, device=device, dtype=dtype))
            d_score = d_score.clone()
            d_score[:, G - 1] = d_score[:, G - 1] + penalidade_ficar
            mover = torch.argmin(d_score, dim=1)

            acao = mover.clone()
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)
            acao = torch.where(segunda, torch.full_like(acao, G + 7), acao)

            self.comeu_ultimo_tick = acao == (G + 7)

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaL9Livre
```
