PREDICAO: mortos ≈0.62–0.66 (perto da base p7+água, ~0.61–0.62); corridas ≈18.0–18.5; refeições ≈7.8–8.1; dano cai um pouco (menos viagens repetidas à oficina), fome estável ~50%.

Lógica: teto de reparo (e_vida<0.8) força cadeias de 2-3 visitas mesmo que a melhor política medida (r7_1, 0.73) nunca empurre vida além de p90=0.726 nos reparos reais. Baixo teto pra 0.70 — uma visita só (entrada <0.5 +0.25 de cura = 0.75) já cruza esse teto quase sempre, cortando viagem extra sem mudar quando entra (H17 já mostrou 2x que entrar mais cedo/mais barato piora, 0.82, corridas caem 16.4→12.7 — aqui não toco a entrada, só o teto de saída). Calor 0.75→0.70: ainda sobra ~22 pts / ~147 ticks até a pane (92) — buffer maior que o original (17/113) — trigger único, sem persistência, não deve gerar desvio extra relevante; só reduz risco de estourar 92 em corrida longa (dano é ~45-48% das mortes junto com fome).

import torch


def fabrica():
    class PoliticaL6Reparo:
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.comeu_ultimo_tick = torch.zeros(m.B, dtype=torch.bool, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # RUA (herdada de p7, intacta): peso pequeno so' desempata direcoes quase
            # equivalentes; L1 p7 usou peso=5.0 e paralisou (0.99, 1.7 corridas).
            PESO_RUA = 0.15
            mover = torch.argmin(F[:, :, 0] + PESO_RUA * F[:, :, 6], dim=1)
            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)

            score = torch.full((B, S), -1e9, device=device, dtype=dtype)

            quer_agua = e_agua < 0.50
            quer_comida = e_fome > 0.15
            quer_fuel = e_fuel < 0.95

            # ORDEM DE COMPRA NO POSTO: comida > agua > fuel (inalterado da base).
            score[:, 7] = torch.where(no_posto & quer_comida, 30.0 + (e_fome - 0.15), score[:, 7])

            idx_pref_agua = torch.where(
                e_agua < 0.25,
                torch.full((B,), 6, device=device, dtype=torch.long),
                torch.full((B,), 5, device=device, dtype=torch.long),
            )
            disp_pref_agua = disp.gather(1, idx_pref_agua.unsqueeze(1)).squeeze(1)
            idx_agua = torch.where(disp_pref_agua, idx_pref_agua, torch.full_like(idx_pref_agua, 5))
            disp_agua = torch.where(disp_pref_agua, disp_pref_agua, disp[:, 5])
            val_agua = torch.where(
                no_posto & quer_agua & disp_agua,
                20.0 + (0.50 - e_agua),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_agua.unsqueeze(1), val_agua.unsqueeze(1))

            # GASOLINA (herdada de p7 intacta): 3 faixas de caixa.
            idx_pref = torch.where(
                dinh > 50,
                torch.full((B,), 3, device=device, dtype=torch.long),
                torch.where(
                    dinh > 25,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.full((B,), 0, device=device, dtype=torch.long),
                ),
            )
            disp_pref = disp.gather(1, idx_pref.unsqueeze(1)).squeeze(1)
            idx_fuel = torch.where(disp_pref, idx_pref, torch.zeros_like(idx_pref))
            disp_fuel = torch.where(disp_pref, disp_pref, disp[:, 0])
            val_fuel = torch.where(
                no_posto & quer_fuel & disp_fuel,
                10.0 + (1.0 - e_fuel),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_fuel.unsqueeze(1), val_fuel.unsqueeze(1))

            melhor = torch.argmax(score, dim=1)
            comprar = no_posto & (score.gather(1, melhor.unsqueeze(1)).squeeze(1) > -1e8)

            segunda = no_posto & self.comeu_ultimo_tick & (e_fome > 0.15) & (dinh > 30) & disp[:, 7]

            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            # ALAVANCA REPARO (unica mudanca vs base): entrada de vida igual (<0.5, dinh>20 —
            # H4 validou esse gatilho, 0.76). TETO de persistencia 0.8 -> 0.70: r7_1 (melhor
            # politica medida, 0.73) so' repara ate vida p90=0.726 na pratica — 0.8 forcava
            # cadeia de 2-3 visitas (25 vida/visita) alem do que qualquer politica boa
            # realmente atinge, gastando ticks/caixa que H17 (2x) mostrou custar corridas
            # (16.4->12.7, 0.82) quando reparo consome demais. Com entrada <0.5, uma visita
            # (+0.25) ja cruza 0.70 quase sempre — corta a viagem extra sem mexer em QUANDO
            # comeca a reparar (nao repete o erro de H17, que testou entrada mais cedo/barata).
            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.70) & (dinh > 10)
            # CALOR 0.75 -> 0.70: gatilho unico, sem persistencia — ainda sobra ~22 pts
            # (~147 ticks a .15/tick) ate a pane de calor (92), buffer MAIOR que o original
            # (17 pts/113 ticks), entao nao deve gerar desvio extra relevante; so' reduz risco
            # de estourar 92 em corrida longa. Dano (pane+calor) e' ~45-48% das mortes, empatado
            # com fome — vale testar isolado do reparo por vida.
            quer_calor = (e_calor > 0.70) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            acao = mover.clone()
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)
            acao = torch.where(segunda, torch.full_like(acao, G + 7), acao)

            self.comeu_ultimo_tick = acao == (G + 7)

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaL6Reparo
