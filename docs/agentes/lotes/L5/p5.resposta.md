PREDICAO: mortos ≈0.72 (leve melhora sobre base 0.73 — alavanca ordem/ticks já historicamente neutra: H19/L1p5 deu 0.74 igual à base, L3p5 com fila explícita quebrou pra 0.99); corridas ≈16.6; refeições ≈7.3; dia ≈5.7; causa fome ~53% / dano ~44% (leve queda no dano: menos ticks gastos completando tanque quase cheio). Sem interrupção de visita no código (comprar sempre domina mover enquanto houver necessidade), reordenar os SCORES de prioridade é no-op — o visitante sempre termina as 3 compras na mesma ordem de ticks independente de qual score é maior. O único jeito real de mexer em "ordem e quantos ticks fica" sem virar máquina de estados (que já quebrou em L3p5) é mudar OS GATILHOS que decidem quantas compras acontecem por visita: (1) limiar de dinheiro pra escolher 100L em vez de 50L, baixado de 40 pra 35 (nota do consolidador, dado explícito); (2) gate de "quer combustível" apertado de <0.95 pra <0.90, cortando o topo da cauda (p90 medido = 88.2% cheio) onde a visita gasta um tick completando um tanque já quase cheio sem necessidade real.

```python
import torch


def fabrica():
    class PoliticaL5Ordem:
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.comeu_ultimo_tick = torch.zeros(m.B, dtype=torch.bool, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # ALAVANCA RUA (herdada de p7, intacta): score de mover ao alvo pondera d_alvo +
            # peso*custo_frente. peso pequeno de proposito: L1 p7 usou peso=5.0 e a penalidade
            # (5*2=10) dominou o d_alvo (diff maxima ~2), paralisando o taxi (0.99, 1.7 corridas).
            # Aqui peso=.15 da penalidade maxima .15*2=.3, so' desempata direcoes quase
            # equivalentes sem nunca virar a decisao principal de ir ao alvo.
            PESO_RUA = 0.15
            mover = torch.argmin(F[:, :, 0] + PESO_RUA * F[:, :, 6], dim=1)
            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)

            score = torch.full((B, S), -1e9, device=device, dtype=dtype)

            quer_agua = e_agua < 0.50
            quer_comida = e_fome > 0.15
            # ALAVANCA ORDEM/TICKS (2/2): gate de fuel apertado de <0.95 pra <0.90. p90 medido
            # de tanque-no-abastecimento da base e' 88.2% cheio — ou seja, 1 em cada 10 visitas
            # a base gasta um tick topando um tanque ja quase cheio (sem risco real: autonomia
            # a 90% ainda tem folga enorme sobre d_posto). Cortar essa cauda tira ticks paraddensimos
            # perto de servico sem comida/agua/combustivel de verdade em jogo — o proprio
            # consolidador mediu 43% de tempo livre no raio de servico na melhor politica vs
            # 26% no campeao, e pediu reduzir esse desperdicio.
            quer_fuel = e_fuel < 0.90

            # ORDEM DE COMPRA NO POSTO: comida > agua > fuel (inalterado — nao e' um lever real
            # aqui: score e' calculado por opcao independente e argmax por tick; sem mecanismo
            # de interrupcao de visita no codigo, a visita sempre completa as 3 compras
            # necessarias em ate 3 ticks nao importa a ordem dos pesos 30/20/10. O que MOVE o
            # resultado e' QUANTAS compras acontecem por visita, nao a ordem entre elas —
            # por isso as duas mudancas reais desta alavanca estao nos gatilhos de fuel abaixo).
            score[:, 7] = torch.where(no_posto & quer_comida, 30.0 + (e_fome - 0.15), score[:, 7])

            idx_pref_agua = torch.where(
                e_agua < 0.25,
                torch.full((B,), 6, device=device, dtype=torch.long),
                torch.full((B,), 5, device=device, dtype=torch.long),
            )
            disp_pref_agua = disp.gather(1, idx_pref_agua.unsqueeze(1)).squeeze(1)
            idx_agua = torch.where(disp_pref_agua, idx_pref_agua, torch.full_like(idx_pref_agua, 5))
            disp_agua = torch.where(disp_pref_agua, disp_pref_agua, disp[:, 5])
            val_agua = torch.where(
                no_posto & quer_agua & disp_agua,
                20.0 + (0.50 - e_agua),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_agua.unsqueeze(1), val_agua.unsqueeze(1))

            # ALAVANCA ORDEM/TICKS (1/2): limiar de dinheiro pra escolher 100L (idx 3) em vez
            # de 50L (idx 1) baixado de 40 pra 35 — pedido explicito do consolidador (nota a).
            # Compra maior = menos viagens dedicadas a fuel depois = menos ticks totais parados
            # no raio de servico ao longo do jogo, mesmo com cada visita individual do mesmo
            # tamanho (<=3 ticks).
            idx_pref = torch.where(
                dinh >= 35,
                torch.full((B,), 3, device=device, dtype=torch.long),
                torch.full((B,), 1, device=device, dtype=torch.long),
            )
            disp_pref = disp.gather(1, idx_pref.unsqueeze(1)).squeeze(1)
            idx_fuel = torch.where(disp_pref, idx_pref, torch.zeros_like(idx_pref))
            disp_fuel = torch.where(disp_pref, disp_pref, disp[:, 0])
            val_fuel = torch.where(
                no_posto & quer_fuel & disp_fuel,
                10.0 + (1.0 - e_fuel),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_fuel.unsqueeze(1), val_fuel.unsqueeze(1))

            melhor = torch.argmax(score, dim=1)
            comprar = no_posto & (score.gather(1, melhor.unsqueeze(1)).squeeze(1) > -1e8)

            # segunda refeicao: se comeu no tick anterior e ainda ta com fome e tem grana, come de novo
            segunda = no_posto & self.comeu_ultimo_tick & (e_fome > 0.15) & (dinh > 30) & disp[:, 7]

            # NOITE: posto fecha de noite -> desvia pro ponto de comida mais perto (raio 25).
            # Nunca desvia com passageiro a bordo (~tem_pax) — corrida termina primeiro.
            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            # emergencia fora do posto: so' se ja esta ao lado (inalterado da base — backstop, nao a alavanca)
            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            # REPARO: (1) vida baixa com histerese .5/.8, (2) calor alto sem passageiro
            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.8) & (dinh > 10)
            quer_calor = (e_calor > 0.75) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            # prioridade (menor pra maior, ultima escrita vence):
            # mover base < buscar comida de longe < detour oficina < compra posto < compra oficina
            # < emergencia agua/comida < segunda refeicao
            acao = mover.clone()
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)
            acao = torch.where(segunda, torch.full_like(acao, G + 7), acao)

            self.comeu_ultimo_tick = acao == (G + 7)

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaL5Ordem
```
