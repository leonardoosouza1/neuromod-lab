PREDICAO: mortos ≈ 0.60–0.68 (piora vs. base 0.53/0.55 — esta é essencialmente a mesma alavanca estrutural do L12 p5, que já mediu 0.66; H40 também já mostrou que capar ticks no posto machuca). Corridas/dia vivo ≈ 2.6–2.9 (caindo do 3.15 da base), refeições caindo de 8.4 para ~6.5–7.5 porque o teto de 2 ticks às vezes deixa fome > 0.3 sem resolver (empurra pro fora_comida de emergência, e_fome>0.85). Não espero empate: cortar a compra de água no posto (threshold 0.50→nada, só sobra o fallback de emergência 0.15) e trocar o tier de gasolina por preço (L12 p4) por um limiar fixo de caixa (35) tende a piorar tanto fome quanto sede, mesmo mantendo resfriamento/reparo/pane-é-transporte intactos.

```python
import torch


def fabrica():
    class PoliticaParadaMinima(object):
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.resfriando = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.ticks_no_posto = torch.zeros(m.B, dtype=torch.long, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # RUA (herdada, intacta): peso pequeno so' desempata direcoes quase equivalentes.
            PESO_RUA = 0.15
            mover = torch.argmin(F[:, :, 0] + PESO_RUA * F[:, :, 6], dim=1)
            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)

            # RESFRIAMENTO (herdado do p8, intacto): histerese e_calor>.667 entra / <=.55 sai.
            entra_zona_sangramento = (e_calor > 0.667) & (~tem_pax)
            self.resfriando = (self.resfriando | entra_zona_sangramento) & (e_calor > 0.55) & (~tem_pax)

            # ALAVANCA PARADA MINIMA (unica mudanca vs p8): H40 mostrou que um teto de ticks
            # QUEBRA o abastecimento quando a propria compra precisa de varios ticks pra se
            # completar. Aqui o ato de comprar continua 1 acao = 1 tick (como o contrato ja
            # exige), o teto so' limita a FILA de itens por visita a no maximo 2: tick1 =
            # gasolina (se precisar) senao ja adianta comida; tick2 = comida (se ainda faltou).
            # Depois disso o taxi SAI mesmo que agua/comida ainda estejam abaixo do limiar --
            # e' o preco explicito da alavanca (nada de agua no posto, "nada mais").
            # self.ticks_no_posto conta ticks CONSECUTIVOS dentro do raio; zera ao sair.
            self.ticks_no_posto = torch.where(
                no_posto, self.ticks_no_posto + 1, torch.zeros_like(self.ticks_no_posto)
            )
            primeiro_tick = no_posto & (self.ticks_no_posto == 1)
            segundo_tick = no_posto & (self.ticks_no_posto == 2)

            quer_fuel = e_fuel < 0.95
            quer_comida = e_fome > 0.3

            idx_fuel = torch.where(
                dinh >= 35.0,
                torch.full((B,), 3, device=device, dtype=torch.long),  # 100 L
                torch.full((B,), 1, device=device, dtype=torch.long),  # 50 L
            )
            disp_fuel = disp.gather(1, idx_fuel.unsqueeze(1)).squeeze(1)

            comprar_fuel = primeiro_tick & quer_fuel & disp_fuel
            comprar_comida_p1 = primeiro_tick & (~quer_fuel) & quer_comida & disp[:, 7]
            comprar_comida_p2 = segundo_tick & quer_comida & disp[:, 7]
            comprar_comida = comprar_comida_p1 | comprar_comida_p2

            # COMIDA NOTURNA (herdada, intacta): posto fecha a noite, ponto de comida nao.
            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            # AGUA/COMIDA de emergencia fora do posto (herdadas, intactas) -- unico canal de
            # agua que sobra, ja' que o posto perdeu a compra proativa de agua nesta alavanca.
            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            # REPARO (herdado, intacto): vida<.5 inicia, so' solta com vida>=.8.
            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.8) & (dinh > 10)
            quer_calor = (e_calor > 0.75) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            parar_resfriar = self.resfriando & (~quer_oficina)

            acao = mover.clone()
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(parar_resfriar, torch.full_like(acao, G - 1), acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar_fuel, G + idx_fuel, acao)
            acao = torch.where(comprar_comida, torch.full_like(acao, G + 7), acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaParadaMinima
```
