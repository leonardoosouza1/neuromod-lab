def fabrica():

            self.comeu_ultimo_tick = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.resfriando = torch.zeros(m.B, dtype=torch.bool, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):

            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_reserva = F[:, 0, 17]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # RUA (herdada de L14 p7, intacta): peso pequeno de custo_frente na
            # escolha de direcao ao alvo, testado 2x (empate em .15, paralisia em 5.0).
            PESO_RUA = 0.3

            # TRABALHO/CORRIDA (alavanca deste lote): so' d_alvo e' visivel (nao da
            # pra mirar um passageiro especifico mais caro), entao a decisao vira
            # "seguir reto ate o mais perto" vs "cacar em outro distrito" usando
            # preco_frente como proxy de fare do proximo passo. So ativa quando
            # ~tem_pax (com passageiro a bordo o destino e' fixo, nada a decidir) E

            PESO_PRECO = 0.2

                caca_preco,

                torch.zeros(B, device=device, dtype=dtype),
            )

            mover = torch.argmin(mover_score, dim=1)

            quer_comida = e_fome > 0.15

                no_posto & quer_agua & disp_agua,
                20.0 + (0.50 - e_agua),
                torch.full((B,), -1e9, device=device, dtype=dtype),

                liters_falta <= 25.0,

                    torch.full((B,), 1, device=device, dtype=torch.long),

            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(parar_resfriar, torch.full_like(acao, G - 1), acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)
            acao = torch.where(segunda, torch.full_like(acao, G + 7), acao)

            self.comeu_ultimo_tick = acao == (G + 7)

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaL16TrabalhoCorridaP6
