Vou implementar a alavanca **NOITE: fadiga/descanso**, de forma bem conservadora — os 3 testes anteriores (H25: L4/L5/L6 p6) falharam por cortar tempo de rodagem à noite (0.74–0.78, corridas caindo pra 14–16). O único teste que empatou com a base (L7 p6 = 0.56) sugere que só funciona se for quase invisível: gatilho raro, nunca disputa com passageiro perto, nunca desvia muito.

Uso `e_fadiga` (coluna 13 de F, hoje totalmente ignorada) e o serviço `descanso` (idx 8), com a mesma distância dedicada `d_descanso` (coluna 5 de NOMES_OP) que já existe pra oficina (`d_oficina`, coluna 4). Copio a base L6 p4 (gasolina) inteira e só acrescento essa lógica, com prioridade mais baixa que tudo — comida, água, oficina e corrida sempre vencem.

PREDICAO: mortos ≈0.55–0.57 (essencialmente empate com a base 0.56 — fadiga não é causa de morte nomeada nos dados, então o efeito direto deve ser perto de zero; o objetivo real deste teste é confirmar se dá pra tocar fadiga sem CUSTAR corrida, não ganhar mortos). corridas ≈18.8–19.0 (gatilho exige `~tem_pax` E passageiro não estar a <15 u E ponto de descanso a <20 u — deve disparar raramente e nunca brigar com uma corrida disponível). refeicoes ≈8.3–8.4 (inalterado). causas: fome ~47%, dano ~47–48%, sede ~20–21% (nenhuma delas é atacada por esta alavanca — é teste de "não piora", não de "melhora").

import torch


def fabrica():
    class PoliticaL6Noite:
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.comeu_ultimo_tick = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.descansando = torch.zeros(m.B, dtype=torch.bool, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_fad = F[:, 0, 13]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]
            no_descanso = disp[:, 8]

            # ALAVANCA RUA (herdada de p7, intacta): score de mover ao alvo pondera d_alvo +
            # peso*custo_frente. peso pequeno de proposito: L1 p7 usou peso=5.0 e a penalidade
            # (5*2=10) dominou o d_alvo (diff maxima ~2), paralisando o taxi (0.99, 1.7 corridas).
            # Aqui peso=.15 da penalidade maxima .15*2=.3, so' desempata direcoes quase
            # equivalentes sem nunca virar a decisao principal de ir ao alvo.
            PESO_RUA = 0.15
            mover = torch.argmin(F[:, :, 0] + PESO_RUA * F[:, :, 6], dim=1)
            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)
            ir_descanso = torch.argmin(F[:, :, 5], dim=1)

            score = torch.full((B, S), -1e9, device=device, dtype=dtype)

            quer_agua = e_agua < 0.50
            quer_comida = e_fome > 0.15
            quer_fuel = e_fuel < 0.95

            # ORDEM DE COMPRA NO POSTO: comida > agua > fuel (inalterado da base).
            score[:, 7] = torch.where(no_posto & quer_comida, 30.0 + (e_fome - 0.15), score[:, 7])

            idx_pref_agua = torch.where(
                e_agua < 0.25,
                torch.full((B,), 6, device=device, dtype=torch.long),
                torch.full((B,), 5, device=device, dtype=torch.long),
            )
            disp_pref_agua = disp.gather(1, idx_pref_agua.unsqueeze(1)).squeeze(1)
            idx_agua = torch.where(disp_pref_agua, idx_pref_agua, torch.full_like(idx_pref_agua, 5))
            disp_agua = torch.where(disp_pref_agua, disp_pref_agua, disp[:, 5])
            val_agua = torch.where(
                no_posto & quer_agua & disp_agua,
                20.0 + (0.50 - e_agua),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_agua.unsqueeze(1), val_agua.unsqueeze(1))

            # ALAVANCA GASOLINA (herdada de L6 p4, intacta): idx_pref = min(tier_por_caixa,
            # tier_por_necessidade), nunca compra mais litro do que cabe no tanque.
            liters_falta = (1.0 - e_fuel) * 100.0
            idx_necessidade = torch.where(
                liters_falta <= 25.0,
                torch.full((B,), 0, device=device, dtype=torch.long),
                torch.where(
                    liters_falta <= 50.0,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.where(
                        liters_falta <= 75.0,
                        torch.full((B,), 2, device=device, dtype=torch.long),
                        torch.full((B,), 3, device=device, dtype=torch.long),
                    ),
                ),
            )
            idx_caixa = torch.where(
                dinh > 50,
                torch.full((B,), 3, device=device, dtype=torch.long),
                torch.where(
                    dinh > 25,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.full((B,), 0, device=device, dtype=torch.long),
                ),
            )
            idx_pref = torch.minimum(idx_caixa, idx_necessidade)
            disp_pref = disp.gather(1, idx_pref.unsqueeze(1)).squeeze(1)
            idx_fuel = torch.where(disp_pref, idx_pref, torch.zeros_like(idx_pref))
            disp_fuel = torch.where(disp_pref, disp_pref, disp[:, 0])
            val_fuel = torch.where(
                no_posto & quer_fuel & disp_fuel,
                10.0 + (1.0 - e_fuel),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_fuel.unsqueeze(1), val_fuel.unsqueeze(1))

            melhor = torch.argmax(score, dim=1)
            comprar = no_posto & (score.gather(1, melhor.unsqueeze(1)).squeeze(1) > -1e8)

            segunda = no_posto & self.comeu_ultimo_tick & (e_fome > 0.15) & (dinh > 30) & disp[:, 7]

            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.8) & (dinh > 10)
            quer_calor = (e_calor > 0.75) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            # ALAVANCA NOITE/FADIGA (unica mudanca vs base): e_fadiga (coluna 13 de F) e' hoje
            # totalmente ignorada. 3 testes anteriores (L4/L5/L6 p6) tentaram "parar de rodar
            # a noite" e cortaram corridas para 14.3-16.0 (mortos 0.74-0.78) sem reduzir fome
            # (H25: aposentar). O 4o teste (L7 p6) usou fadiga sem esvaziar as horas noturnas e
            # empatou com a base (0.56) -- indicio de que so' funciona se for quase invisivel.
            # Aqui o gatilho e' deliberadamente raro e de prioridade minima: so' entra com
            # fadiga quase no teto (>.85, contra "hoje ignorado" = nunca), sem passageiro a
            # bordo NEM por perto (F[:,G-1,0]>15 = ninguem a <15u, pra nunca perder uma corrida
            # disponivel por causa de descanso), caixa > 15 (hotel custa 8) e ponto de descanso
            # a <20u (nunca um desvio longo, ao contrario das tentativas anteriores). Historese
            # (self.descansando) evita entrar e sair do servico em ticks alternados: uma vez
            # comecado, so' sai com fadiga < .45 ou caixa curta ou passageiro surgindo perto.
            gatilho_descanso = (
                (e_fad > 0.85)
                & (~tem_pax)
                & (dinh > 15)
                & (F[:, G - 1, 5] < 20)
                & (F[:, G - 1, 0] > 15)
            )
            self.descansando = (
                (self.descansando | gatilho_descanso)
                & (e_fad > 0.45)
                & (dinh > 8)
                & (~tem_pax)
                & (F[:, G - 1, 0] > 15)
            )
            comprar_descanso = no_descanso & self.descansando
            ir_descanso_mov = self.descansando & (~comprar_descanso)

            acao = mover.clone()
            acao = torch.where(ir_descanso_mov, ir_descanso, acao)
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_descanso, torch.full_like(acao, G + 8), acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)
            acao = torch.where(segunda, torch.full_like(acao, G + 7), acao)

            self.comeu_ultimo_tick = acao == (G + 7)

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaL6Noite
