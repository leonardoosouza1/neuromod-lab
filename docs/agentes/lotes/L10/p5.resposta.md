PREDICAO: mortos ≈0.54–0.60 (aposta neutra a levemente favorável — a alavanca ataca só o *tempo de parada*, não os limiares que já saturaram; risco declarado: como a ordem de prioridade fica comida>água>gasolina e o teto agora é rígido de 2 ticks, nas visitas em que fome, sede e tanque baixo coincidem, a gasolina — motivo real da parada — pode ficar de fora e empurrar panes/corridas para baixo; se isso dominar, mortos vai para ~0.60–0.64). Corridas/dia vivo ≈3.2–3.6 (base 3.28); dia vivo ≈5.7–6.0.

O bug do L8 p5 (0.94 mortos, 2,9 abastecidas, 9,8 panes) não foi limiar — foi trava: a versão anterior só saía do raio quando toda compra pendente estivesse resolvida, e quando alguma delas ficava indisponível (ex.: tier de litro fora de alcance do caixa) o carro nunca zerava a necessidade e nunca saía. Corrijo isso com um contador `ticks_posto` que força saída incondicional após 2 ticks, independente de compra pendente — a garantia estrutural que a alavanca pede. Mantive a ordem de compra da base (comida > água > gasolina, já validada em H24) e a lógica de água intacta ("compras que já funcionam"); troquei só o gatilho de comida (.15→.3, como pedido) e o cálculo de litro (antes por necessidade do tanque, agora fixo 100 L com dinheiro ≥35 senão 50 L, como pedido) — isso elimina o recálculo por litro faltante e faz cada decisão caber em exatamente 1 tick. Removi a lógica "segunda refeição" (comeu_ultimo_tick) porque ela dependia de loitering de 3+ ticks, incompatível com o teto de 2. Resto (RUA peso .15, reparo com vida<.5, calor, comida noturna fora do posto, panes=transporte) copiado sem alteração.

import torch


def fabrica():
    class PoliticaL10ParadaMinima:
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.no_posto_ultimo = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.ticks_posto = torch.zeros(m.B, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # ALAVANCA RUA (herdada, intacta): peso pequeno so' desempata direcoes
            # quase equivalentes, nunca vira a decisao principal de ir ao alvo.
            PESO_RUA = 0.15
            mover = torch.argmin(F[:, :, 0] + PESO_RUA * F[:, :, 6], dim=1)
            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)

            # PARADA MINIMA (unica mudanca vs base): contador de ticks parado no
            # raio do posto. Zera ao entrar; depois de 2 ticks o bloco de compra
            # abaixo eh desligado (score forcado a -1e9) mesmo com necessidade
            # pendente — saida incondicional. Isso evita o travamento medido em
            # L8 p5 (0.94 mortos, 2.9 abastecidas, 9.8 panes): la' o taxi ficava
            # preso tentando resolver toda compra antes de sair; aqui o teto e'
            # estrutural, nao depende de nenhuma necessidade zerar.
            entrando = no_posto & (~self.no_posto_ultimo)
            self.ticks_posto = torch.where(entrando, torch.zeros_like(self.ticks_posto), self.ticks_posto)
            self.ticks_posto = torch.where(no_posto, self.ticks_posto + 1.0, torch.zeros_like(self.ticks_posto))
            dentro_janela = no_posto & (self.ticks_posto <= 2.0)

            score = torch.full((B, S), -1e9, device=device, dtype=dtype)

            # ORDEM DE COMPRA (igual a base: comida > agua > gasolina, H24). So'
            # dispara dentro_janela (<=2 ticks). Comida sobe de fome>.15 para
            # fome>.3 e a gasolina vira regra fixa por caixa (100 L com dinheiro
            # >= 35 — mediana do campeao e' 37.1 — senao 50 L) em vez do calculo
            # por litro faltante da base: decisao mais simples, cabe em 1 tick
            # cada, sem recalculo de tier por necessidade do tanque.
            quer_comida = e_fome > 0.3
            quer_agua = e_agua < 0.50
            quer_fuel = e_fuel < 0.95

            score[:, 7] = torch.where(dentro_janela & quer_comida, 30.0 + (e_fome - 0.3), score[:, 7])

            idx_pref_agua = torch.where(
                e_agua < 0.25,
                torch.full((B,), 6, device=device, dtype=torch.long),
                torch.full((B,), 5, device=device, dtype=torch.long),
            )
            disp_pref_agua = disp.gather(1, idx_pref_agua.unsqueeze(1)).squeeze(1)
            idx_agua = torch.where(disp_pref_agua, idx_pref_agua, torch.full_like(idx_pref_agua, 5))
            disp_agua = torch.where(disp_pref_agua, disp_pref_agua, disp[:, 5])
            val_agua = torch.where(
                dentro_janela & quer_agua & disp_agua,
                20.0 + (0.50 - e_agua),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_agua.unsqueeze(1), val_agua.unsqueeze(1))

            idx_fuel = torch.where(
                dinh >= 35.0,
                torch.full((B,), 3, device=device, dtype=torch.long),
                torch.full((B,), 1, device=device, dtype=torch.long),
            )
            disp_fuel = disp.gather(1, idx_fuel.unsqueeze(1)).squeeze(1)
            val_fuel = torch.where(
                dentro_janela & quer_fuel & disp_fuel,
                10.0 + (1.0 - e_fuel),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_fuel.unsqueeze(1), val_fuel.unsqueeze(1))

            melhor = torch.argmax(score, dim=1)
            comprar = no_posto & (score.gather(1, melhor.unsqueeze(1)).squeeze(1) > -1e8)

            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.8) & (dinh > 10)
            quer_calor = (e_calor > 0.75) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            acao = mover.clone()
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)

            self.no_posto_ultimo = no_posto.clone()

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaL10ParadaMinima
