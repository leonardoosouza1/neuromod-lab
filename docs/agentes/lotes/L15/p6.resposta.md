Espero empate: mortos ≈ 0.51–0.56 (base 0.52), corridas/dia vivo levemente abaixo (2.20–2.35 vs 2.32, por causa de detours ocasionais), fome/dano sem mudança relevante (a alavanca não toca nutrição nem calor). Se sair fora desse intervalo, é porque o gate por distância muda a dinâmica de forma que o peso cego não capturava.

```python
import torch


def fabrica():
    class PoliticaL15TrabalhoCorrida(object):
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.comeu_ultimo_tick = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.resfriando = torch.zeros(m.B, dtype=torch.bool, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # RUA (herdada de L13/L14 p7, intacta): peso pequeno de custo_frente na
            # escolha de direcao ao alvo. Empatou 2x (0.53/0.52) contra base sem peso;
            # mantido porque nao piora e reduz desgaste marginal andando fora da rua.
            PESO_RUA = 0.3

            # ALAVANCA DESTE LOTE (TRABALHO/CORRIDA): unico eixo que ainda nao tinha
            # sido testado como GATE condicional -- as duas tentativas anteriores
            # (L14 p6 e uma antes, H55) aplicaram peso de preco_frente CEGO, o tempo
            # todo, inclusive levando o taxi vazio ('~tem_pax') a desviar mesmo quando
            # o passageiro mais perto ja estava a poucas unidades (d_alvo baixo) --
            # nesse regime, o detour custa mais corrida do que ganha em tarifa, e por
            # isso mediu empate com a base (0.53): o efeito positivo (achar corrida
            # mais cara) e o negativo (perder a corrida perto) se cancelam.
            #
            # Aqui o peso de preco_frente so' entra quando (a) o taxi esta' LIVRE
            # (~tem_pax: com passageiro a bordo, F[:,:,0] ja' e' d_alvo=distancia ao
            # DESTINO da corrida atual -- nao ha' "outro distrito" a escolher, seguir
            # reto e' estritamente melhor, dado que so' se recebe ao entregar) e (b) o
            # alvo mais proximo ja' esta' longe mesmo parado (F[:,G-1,0] > LIMIAR_LONGE
            # = 15 unidades -- mediana de d_posto na pane e' 25, e a mediana de
            # intervalo entre corridas do campeao e' 57 ticks, entao 15 unidades e' um
            # raio conservador que so' classifica "vou caminhar de qualquer jeito").
            # So' nesse regime vale gastar o mesmo deslocamento perseguindo o rumo mais
            # caro (preco_frente maior) em vez do rumo mais raso ao alvo: o custo de
            # oportunidade do detour e' quase zero porque o passageiro mais perto ja'
            # nao estava ali do lado.
            #
            # PESO_PRECO = 0.2, abaixo do teto de 0.3 recomendado pelo consolidador
            # para pesos de direcao (mesma ordem de grandeza do PESO_RUA ja' validado
            # como nao-catastrofico): a diferenca tipica de d_alvo entre duas direcoes
            # e' da ordem de poucas unidades ate' dezenas, entao 0.2*diff_preco so'
            # vira decisivo quando a diferenca de preco entre rumos e' proporcionalmente
            # grande -- nao um degrau que paralisa como o peso 5.0 original de custo.
            LIMIAR_LONGE = 15.0
            PESO_PRECO = 0.2

            alvo_livre_longe = (~tem_pax) & (F[:, G - 1, 0] > LIMIAR_LONGE)

            mover_reto = torch.argmin(F[:, :, 0] + PESO_RUA * F[:, :, 6], dim=1)
            mover_cacar = torch.argmin(
                F[:, :, 0] + PESO_RUA * F[:, :, 6] - PESO_PRECO * F[:, :, 7], dim=1
            )
            mover = torch.where(alvo_livre_longe, mover_cacar, mover_reto)

            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)

            # LIVRE / CALOR (herdada de L12 p8, intacta).
            entra_zona_sangramento = (e_calor > 0.667) & (~tem_pax)
            self.resfriando = (self.resfriando | entra_zona_sangramento) & (e_calor > 0.55) & (~tem_pax)

            score = torch.full((B, S), -1e9, device=device, dtype=dtype)

            quer_agua = e_agua < 0.50
            quer_comida = e_fome > 0.15
            quer_fuel = e_fuel < 0.95

            score[:, 7] = torch.where(no_posto & quer_comida, 30.0 + (e_fome - 0.15), score[:, 7])

            idx_pref_agua = torch.where(
                (e_agua < 0.25) & (dinh > 40.0),
                torch.full((B,), 6, device=device, dtype=torch.long),
                torch.full((B,), 5, device=device, dtype=torch.long),
            )
            disp_pref_agua = disp.gather(1, idx_pref_agua.unsqueeze(1)).squeeze(1)
            idx_agua = torch.where(disp_pref_agua, idx_pref_agua, torch.full_like(idx_pref_agua, 5))
            disp_agua = torch.where(disp_pref_agua, disp_pref_agua, disp[:, 5])
            val_agua = torch.where(
                no_posto & quer_agua & disp_agua,
                20.0 + (0.50 - e_agua),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_agua.unsqueeze(1), val_agua.unsqueeze(1))

            liters_falta = (1.0 - e_fuel) * 100.0
            idx_necessidade = torch.where(
                liters_falta <= 25.0,
                torch.full((B,), 0, device=device, dtype=torch.long),
                torch.where(
                    liters_falta <= 50.0,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.where(
                        liters_falta <= 75.0,
                        torch.full((B,), 2, device=device, dtype=torch.long),
                        torch.full((B,), 3, device=device, dtype=torch.long),
                    ),
                ),
            )
            idx_caixa = torch.where(
                dinh > 50,
                torch.full((B,), 3, device=device, dtype=torch.long),
                torch.where(
                    dinh > 25,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.full((B,), 0, device=device, dtype=torch.long),
                ),
            )
            idx_pref = torch.minimum(idx_caixa, idx_necessidade)
            disp_pref = disp.gather(1, idx_pref.unsqueeze(1)).squeeze(1)
            idx_fuel = torch.where(disp_pref, idx_pref, torch.zeros_like(idx_pref))
            disp_fuel = torch.where(disp_pref, disp_pref, disp[:, 0])
            val_fuel = torch.where(
                no_posto & quer_fuel & disp_fuel,
                10.0 + (1.0 - e_fuel),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_fuel.unsqueeze(1), val_fuel.unsqueeze(1))

            melhor = torch.argmax(score, dim=1)
            comprar = no_posto & (score.gather(1, melhor.unsqueeze(1)).squeeze(1) > -1e8)

            segunda = no_posto & self.comeu_ultimo_tick & (e_fome > 0.15) & (dinh > 30) & disp[:, 7]

            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.8) & (dinh > 10)
            quer_calor = (e_calor > 0.75) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            parar_resfriar = self.resfriando & (~quer_oficina)

            acao = mover.clone()
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(parar_resfriar, torch.full_like(acao, G - 1), acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)
            acao = torch.where(segunda, torch.full_like(acao, G + 7), acao)

            self.comeu_ultimo_tick = acao == (G + 7)

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaL15TrabalhoCorrida
```
