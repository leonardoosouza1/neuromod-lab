PREDICAO: mortos ≈ 0.49–0.53 (melhora modesta sobre a base 0.53 — a sede já responde por ~32% das mortes em L13 e nenhuma das 4 rodadas anteriores de "água: quando e quanto" tocou o fato de que o posto fecha à noite mas o ponto de comida vende água o tempo todo; threshold de gatilho igual ao campeão ~0.55, só abre o canal noturno que faltava); corridas/dia vivo ≈ 3.00–3.15 (quase igual à base 3.15 — o desvio de água à noite normalmente coincide com o desvio de comida já existente, custo marginal baixo).

```python
import torch


def fabrica():
    class PoliticaAguaNoturna(object):
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.comeu_ultimo_tick = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.resfriando = torch.zeros(m.B, dtype=torch.bool, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # ALAVANCA RUA (herdada de p7, intacta): score de mover ao alvo pondera d_alvo +
            # peso*custo_frente. peso pequeno de proposito: L1 p7 usou peso=5.0 e a penalidade
            # (5*2=10) dominou o d_alvo (diff maxima ~2), paralisando o taxi (0.99, 1.7 corridas).
            # Aqui peso=.15 da penalidade maxima .15*2=.3, so' desempata direcoes quase
            # equivalentes sem nunca virar a decisao principal de ir ao alvo.
            PESO_RUA = 0.15
            mover = torch.argmin(F[:, :, 0] + PESO_RUA * F[:, :, 6], dim=1)
            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)

            # RESFRIAMENTO (herdado de L12 p8, intacto): entra na zona de sangramento
            # (D_RODAR .15/tick andando com calor>80 raw = e_calor>.667) e para (ficar) ate'
            # soltar em e_calor<=.55 (raw 66), nunca com passageiro.
            entra_zona_sangramento = (e_calor > 0.667) & (~tem_pax)
            self.resfriando = (self.resfriando | entra_zona_sangramento) & (e_calor > 0.55) & (~tem_pax)

            score = torch.full((B, S), -1e9, device=device, dtype=dtype)

            quer_agua = e_agua < 0.50
            quer_comida = e_fome > 0.15
            quer_fuel = e_fuel < 0.95

            # ORDEM DE COMPRA NO POSTO: comida > agua > fuel (inalterado da base).
            score[:, 7] = torch.where(no_posto & quer_comida, 30.0 + (e_fome - 0.15), score[:, 7])

            # AGUA NO POSTO (inalterada da base): gatilho e_agua<0.50, tier 50L salvo
            # e_agua<0.25 -> 100L.
            idx_pref_agua = torch.where(
                (e_agua < 0.25) & (dinh > 40.0),
                torch.full((B,), 6, device=device, dtype=torch.long),
                torch.full((B,), 5, device=device, dtype=torch.long),
            )
            disp_pref_agua = disp.gather(1, idx_pref_agua.unsqueeze(1)).squeeze(1)
            idx_agua = torch.where(disp_pref_agua, idx_pref_agua, torch.full_like(idx_pref_agua, 5))
            disp_agua = torch.where(disp_pref_agua, disp_pref_agua, disp[:, 5])
            val_agua = torch.where(
                no_posto & quer_agua & disp_agua,
                20.0 + (0.50 - e_agua),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_agua.unsqueeze(1), val_agua.unsqueeze(1))

            # GASOLINA (inalterada da base): tier = min(tier_por_caixa, tier_por_necessidade).
            liters_falta = (1.0 - e_fuel) * 100.0
            idx_necessidade = torch.where(
                liters_falta <= 25.0,
                torch.full((B,), 0, device=device, dtype=torch.long),
                torch.where(
                    liters_falta <= 50.0,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.where(
                        liters_falta <= 75.0,
                        torch.full((B,), 2, device=device, dtype=torch.long),
                        torch.full((B,), 3, device=device, dtype=torch.long),
                    ),
                ),
            )
            idx_caixa = torch.where(
                dinh > 50,
                torch.full((B,), 3, device=device, dtype=torch.long),
                torch.where(
                    dinh > 25,
                    torch.full((B,), 1, device=device, dtype=torch.long),
                    torch.full((B,), 0, device=device, dtype=torch.long),
                ),
            )
            idx_pref = torch.minimum(idx_caixa, idx_necessidade)
            disp_pref = disp.gather(1, idx_pref.unsqueeze(1)).squeeze(1)
            idx_fuel = torch.where(disp_pref, idx_pref, torch.zeros_like(idx_pref))
            disp_fuel = torch.where(disp_pref, disp_pref, disp[:, 0])
            val_fuel = torch.where(
                no_posto & quer_fuel & disp_fuel,
                10.0 + (1.0 - e_fuel),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_fuel.unsqueeze(1), val_fuel.unsqueeze(1))

            melhor = torch.argmax(score, dim=1)
            comprar = no_posto & (score.gather(1, melhor.unsqueeze(1)).squeeze(1) > -1e8)

            segunda = no_posto & self.comeu_ultimo_tick & (e_fome > 0.15) & (dinh > 30) & disp[:, 7]

            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            # ALAVANCA AGUA (unica mudanca vs a base): FATO B do consolidador -- o ponto de
            # COMIDA tambem vende agua (serv_ab: perto['agua'] recebe comida e posto) e so' o
            # POSTO fecha a noite. A logica de posto acima ja cobre o dia; de noite o taxi fica
            # sem canal de agua ate' a emergencia (e_agua<0.15 abaixo). Isso deixa "sede" como
            # ~32% das causas de morte no lote anterior sem nenhuma politica ter atacado o
            # mecanismo (so' o limiar no posto, ja' testado 2x em empate: L10 p2, L13 p2).
            # Aqui o gatilho e' o MESMO da base para o posto (agua<0.50, +.05 de folga = .55,
            # igual ao valor medido do proprio campeao evoluido: BEBE com agua mediana 54.5-54.9
            # -- nao inventamos limiar novo, so' abrimos o canal que faltava a noite), ativo so'
            # a noite (e_noite>.5) e sem passageiro. Vai ativamente ate' o ponto de comida
            # (d_comida = F[:,G-1,2], ja' usado por buscar_comida) em vez do d_agua generico,
            # porque e' o unico lugar com agua disponivel de noite; comida tem prioridade
            # quando os dois gatilhos coincidem (buscar_agua_noite so' vale se ~buscar_comida),
            # espelhando a ordem existente comida>agua>fuel no posto.
            quer_agua_noite = (e_agua < 0.55) & (e_noite > 0.5) & (~tem_pax) & (~no_posto)
            buscar_agua_noite = quer_agua_noite & (F[:, G - 1, 2] < 25) & (~buscar_comida)

            # tier: 50L (G+5) por padrao, 100L (G+6) se a agua ja esta bem baixa (<.3), com
            # fallback pro tier disponivel -- mesmo padrao da compra de agua no posto acima.
            idx_pref_agua_noite = torch.where(
                e_agua < 0.3,
                torch.full((B,), 6, device=device, dtype=torch.long),
                torch.full((B,), 5, device=device, dtype=torch.long),
            )
            disp_pref_agua_noite = disp.gather(1, idx_pref_agua_noite.unsqueeze(1)).squeeze(1)
            idx_agua_noite = torch.where(disp_pref_agua_noite, idx_pref_agua_noite, torch.full_like(idx_pref_agua_noite, 5))
            fora_agua_noite = quer_agua_noite & (F[:, G - 1, 2] < 6) & (disp_pref_agua_noite | disp[:, 5])

            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.8) & (dinh > 10)
            quer_calor = (e_calor > 0.75) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            parar_resfriar = self.resfriando & (~quer_oficina)

            acao = mover.clone()
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(buscar_agua_noite, ir_comida_mov, acao)
            acao = torch.where(parar_resfriar, torch.full_like(acao, G - 1), acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_agua_noite, G + idx_agua_noite, acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)
            acao = torch.where(segunda, torch.full_like(acao, G + 7), acao)

            self.comeu_ultimo_tick = acao == (G + 7)

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaAguaNoturna
```
