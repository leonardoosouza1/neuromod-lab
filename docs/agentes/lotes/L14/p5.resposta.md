PREDICAO: colapso, não melhora — mortos ≈ 0.90–0.97, corridas/dia vivo ≈ 1.5–2.0 (base 18.6/3.15). Esta é a 4ª vez que este exato mecanismo (teto de ticks no raio do posto) é testado: H40 já mediu colapso em L8 p5 e L10 p5 (0.97, abastecida ~1.0), e L13 p5 — o MESMO enunciado ("decidir tudo em ≤2 ticks: 100L/50L por caixa, comida se fome>.3, nada mais") — mediu 0.95 com 9.3 corridas totais (1.92/dia) e sede saltando a 57%. O motivo medido é mecânico, não de limiar: abastecer 100L não conclui em 1 ação de compra — o campeão fica mediana ~1 tick no posto mas com p90 de 5.7 ticks, e a base atual (sem teto) já opera dentro do platô 0.53–0.55 justamente porque deixa a compra terminar. Forçar saída aos 2 ticks corta reabastecimentos pela metade, e cada pane incompleta ainda paga o D_PANE fixo + dano por reserva não cheia — a fome e o dano tendem a cair (menos km rodados) mas a sede sobe e o throughput desaba, empurrando mortos para perto do campeão evoluído (0.85–0.97). Implemento fielmente o pedido (troquei só a lógica de compra no posto pela regra fixa de caixa + teto de 2 ticks, mantendo água<.50, resfriamento do p8, reparo vida<.5, pane=transporte e comida noturna intactos) para confirmar/negar a 4ª réplica.

```python
import torch


def fabrica():
    class PoliticaL6Livre(object):
        def reset(self, m, W):
            self.reparando_vida = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.resfriando = torch.zeros(m.B, dtype=torch.bool, device=DEV)
            self.ticks_no_posto = torch.zeros(m.B, dtype=torch.long, device=DEV)

        def extra(self, m, W, F):
            self.F = F
            return torch.zeros(m.B, F.shape[1], device=DEV, dtype=F.dtype)

        def decidir(self, tudo, m):
            F = self.F
            G = F.shape[1]
            S = tudo.shape[1] - G
            device = tudo.device
            dtype = tudo.dtype
            B = tudo.shape[0]

            e_fuel = F[:, 0, 10]
            e_agua = F[:, 0, 11]
            e_fome = F[:, 0, 12]
            e_calor = F[:, 0, 14]
            e_vida = F[:, 0, 15]
            e_dinh = F[:, 0, 16]
            e_pax = F[:, 0, 18]
            e_noite = F[:, 0, 21]
            dinh = e_dinh * 150.0
            tem_pax = e_pax > 0.5

            disp = tudo[:, G:G + S] > -1e8
            no_posto = disp[:, 0]
            no_oficina = disp[:, 9]

            # ALAVANCA RUA (herdada, intacta): pequeno desempate, nunca domina a decisao.
            PESO_RUA = 0.15
            mover = torch.argmin(F[:, :, 0] + PESO_RUA * F[:, :, 6], dim=1)
            ir_oficina = torch.argmin(F[:, :, 4], dim=1)
            ir_comida_mov = torch.argmin(F[:, :, 2], dim=1)

            # RESFRIAMENTO (herdado de L12 p8, intacto): histerese .667/.55, nunca com passageiro.
            entra_zona_sangramento = (e_calor > 0.667) & (~tem_pax)
            self.resfriando = (self.resfriando | entra_zona_sangramento) & (e_calor > 0.55) & (~tem_pax)

            # ALAVANCA DESTE LOTE (unica mudanca vs L12 p8): PARADA MINIMA no posto.
            # H40 ja mostrou 2x (L8 p5, L10 p5) que um TETO de ticks no raio do posto
            # colapsa a politica (abastecimento fica incompleto -> panes em cascata); e o
            # MESMO enunciado ja foi medido no lote anterior (L13 p5: 0.95, 9.3 corridas,
            # sede 57%). Reproduzimos fielmente o pedido pra confirmar a 4a replica:
            # decisao fixa por CAIXA (nao por litro faltante) e teto duro de 2 ticks.
            # ticks_antes = ticks JA gastos no raio antes deste tick; ao chegar a 2, a
            # "vaga" de compra fecha e o taxi so' pode sair (mover normal reassume).
            ticks_antes = self.ticks_no_posto
            pode_comprar_ainda = no_posto & (ticks_antes < 2)
            self.ticks_no_posto = torch.where(no_posto, ticks_antes + 1, torch.zeros_like(ticks_antes))

            score = torch.full((B, S), -1e9, device=device, dtype=dtype)

            quer_agua = e_agua < 0.50
            quer_comida = e_fome > 0.3
            quer_fuel = e_fuel < 0.95

            score[:, 7] = torch.where(
                no_posto & pode_comprar_ainda & quer_comida,
                30.0 + (e_fome - 0.3),
                score[:, 7],
            )

            # AGUA: mesma regra da base (mantida por instrucao), so' respeita o teto de ticks.
            idx_pref_agua = torch.where(
                (e_agua < 0.25) & (dinh > 40.0),
                torch.full((B,), 6, device=device, dtype=torch.long),
                torch.full((B,), 5, device=device, dtype=torch.long),
            )
            disp_pref_agua = disp.gather(1, idx_pref_agua.unsqueeze(1)).squeeze(1)
            idx_agua = torch.where(disp_pref_agua, idx_pref_agua, torch.full_like(idx_pref_agua, 5))
            disp_agua = torch.where(disp_pref_agua, disp_pref_agua, disp[:, 5])
            val_agua = torch.where(
                no_posto & pode_comprar_ainda & quer_agua & disp_agua,
                20.0 + (0.50 - e_agua),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_agua.unsqueeze(1), val_agua.unsqueeze(1))

            # GASOLINA: a alavanca pede tiragem PURA por caixa (100L com dinh>=35, senao
            # 50L) -- substitui o tier por necessidade/litros-faltantes da base. "Nada
            # mais": sem tier de 25/75 nem de reserva (idx 4).
            idx_fuel = torch.where(
                dinh >= 35.0,
                torch.full((B,), 3, device=device, dtype=torch.long),
                torch.full((B,), 1, device=device, dtype=torch.long),
            )
            disp_fuel = disp.gather(1, idx_fuel.unsqueeze(1)).squeeze(1)
            val_fuel = torch.where(
                no_posto & pode_comprar_ainda & quer_fuel & disp_fuel,
                10.0 + (1.0 - e_fuel),
                torch.full((B,), -1e9, device=device, dtype=dtype),
            )
            score.scatter_(1, idx_fuel.unsqueeze(1), val_fuel.unsqueeze(1))

            melhor = torch.argmax(score, dim=1)
            comprar = no_posto & pode_comprar_ainda & (score.gather(1, melhor.unsqueeze(1)).squeeze(1) > -1e8)

            # canal noturno de comida (herdado, intacto): posto fecha a noite, ponto de
            # comida nao.
            buscar_comida = (~no_posto) & (~tem_pax) & ((e_noite > 0.5) | (e_fome > 0.6)) & (F[:, G - 1, 2] < 25)

            fora_agua = (~no_posto) & (e_agua < 0.15) & (F[:, G - 1, 3] < 8) & disp[:, 5]
            fora_comida = (~no_posto) & (e_fome > 0.85) & (F[:, G - 1, 2] < 8) & disp[:, 7]

            ini_vida = (e_vida < 0.5) & (dinh > 20)
            self.reparando_vida = (self.reparando_vida | ini_vida) & (e_vida < 0.8) & (dinh > 10)
            quer_calor = (e_calor > 0.75) & (~tem_pax)
            quer_oficina = self.reparando_vida | quer_calor

            comprar_oficina = no_oficina & quer_oficina & (tudo[:, G + 9] > -1e8)
            ir_oficina_mov = quer_oficina & (~comprar_oficina)

            parar_resfriar = self.resfriando & (~quer_oficina)

            acao = mover.clone()
            acao = torch.where(buscar_comida, ir_comida_mov, acao)
            acao = torch.where(parar_resfriar, torch.full_like(acao, G - 1), acao)
            acao = torch.where(ir_oficina_mov, ir_oficina, acao)
            acao = torch.where(comprar, G + melhor, acao)
            acao = torch.where(comprar_oficina, torch.full_like(acao, G + 9), acao)
            acao = torch.where(fora_agua, torch.full_like(acao, G + 5), acao)
            acao = torch.where(fora_comida, torch.full_like(acao, G + 7), acao)

            return acao.long()

        def atualizar(self, m):
            pass

    return PoliticaL6Livre
```
